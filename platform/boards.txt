##############################################################
#  Moovma Elyssa - ESP32-S3FN8 (8 MB embedded flash, no PSRAM)
#
#  USB: two modes, selected in the Tools menu.
#   MODE 1 (default, everyday): USB-OTG (TinyUSB)
#     Windows and the IDE show the board as "Elyssa" (303A:81FF).
#   MODE 2 (debug): Hardware CDC and JTAG
#     Change BOTH menus:  USB Mode    -> Hardware CDC and JTAG
#                         Upload Mode -> UART0 / Hardware CDC
#     Windows shows "USB JTAG/serial debug unit" (303A:1001), the IDE shows "Elyssa".
#     Boot logs, crash messages and JTAG debugging are available.
#
#  RULE: in every menu, the FIRST option listed is the default.
#  RULE: menu option keys (e.g. "8M", "cdc") are a contract:
#        renaming/removing one breaks users who had it selected.
##############################################################

# JTAG Adapter: this menu has NO effect with core 3.3.x. The IDE's Debug
# button always uses the built-in USB JTAG (debug_script.esp32s3 in
# platform.txt). Kept only because option keys are a contract.
# Debugging requires MODE 2 (Hardware CDC and JTAG), plus the tools
# openocd-esp32 and xtensa-esp-elf-gdb (not yet declared in the index).
# External FTDI / bridge removed: JTAG pins are not available on Elyssa
# (GPIO39 = red LED on v5, GPIO40-42 not routed).
elyssa.menu.JTAGAdapter.default=Disabled
elyssa.menu.JTAGAdapter.default.build.copy_jtag_files=0
elyssa.menu.JTAGAdapter.builtin=Integrated USB JTAG
